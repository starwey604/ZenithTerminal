find_package(PkgConfig REQUIRED)
# Native driver integration uses the host FFmpeg ABI; it never enters RK sysroot builds.
pkg_check_modules(AVCODEC REQUIRED IMPORTED_TARGET libavcodec>=60 libavutil>=58)
pkg_check_modules(AVFORMAT REQUIRED IMPORTED_TARGET libavformat>=60)
add_library(zenith_decoder_vaapi STATIC vaapi.cpp)
target_link_libraries(zenith_decoder_vaapi PUBLIC zenith_video PRIVATE PkgConfig::AVCODEC)
target_compile_options(zenith_decoder_vaapi PRIVATE -Wall -Wextra -Wpedantic)
add_library(zenith_video_file STATIC ../../video/video_file.cpp)
target_link_libraries(zenith_video_file PUBLIC zenith_video PRIVATE PkgConfig::AVFORMAT PkgConfig::AVCODEC)
add_executable(zenith_video_replay ../../tools/video_replay.cpp)
target_link_libraries(zenith_video_replay PRIVATE zenith_video_file zenith_backend)
if(ZT_OUTPUT_DIR)
    set_target_properties(zenith_video_replay PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${ZT_OUTPUT_DIR}")
endif()
if(BUILD_TESTING)
    add_executable(zenith_video_file_tests ${PROJECT_SOURCE_DIR}/tests/video_file_tests.cpp)
    target_link_libraries(zenith_video_file_tests PRIVATE zenith_video_file)
    add_test(NAME video-file COMMAND zenith_video_file_tests ${PROJECT_SOURCE_DIR}/assets/video/demo-1080p60.mp4)
    # Explicit hardware opt-in, not a requirement for ordinary CTest.
    add_executable(zenith_video_decode_test ${PROJECT_SOURCE_DIR}/tests/video_decode.cpp)
    target_link_libraries(zenith_video_decode_test PRIVATE zenith_video_file zenith_decoder_vaapi)
endif()
