find_package(PkgConfig REQUIRED)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm>=2.4.110)
add_library(zenith_display_drm STATIC probe.cpp formats.cpp report.cpp atomic_display.cpp)
target_include_directories(zenith_display_drm PUBLIC include)
target_link_libraries(zenith_display_drm PRIVATE PkgConfig::DRM PUBLIC zenith_video)
target_compile_options(zenith_display_drm PRIVATE -Wall -Wextra -Wpedantic)
add_executable(zenith_drm_probe ../../tools/drm_probe.cpp)
target_link_libraries(zenith_drm_probe PRIVATE zenith_display_drm)
if(ZT_OUTPUT_DIR)
    set_target_properties(zenith_drm_probe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${ZT_OUTPUT_DIR}")
endif()
install(TARGETS zenith_drm_probe RUNTIME DESTINATION bin)
if(BUILD_TESTING)
    # Explicit opt-in hardware test; never claim DRM master during normal CTest.
    add_executable(zenith_drm_lifecycle_test ../../../tests/drm_lifecycle.cpp)
    target_link_libraries(zenith_drm_lifecycle_test PRIVATE zenith_display_drm)
    add_executable(zenith_tests ../../../tests/drm_tests.cpp ../../../tests/ui_tests.cpp)
    target_link_libraries(zenith_tests PRIVATE zenith_display_drm zenith_ui PkgConfig::DRM)
    if(NOT CMAKE_CROSSCOMPILING)
        add_test(NAME drm-unit COMMAND zenith_tests)
        add_test(NAME drm-probe-help COMMAND zenith_drm_probe --help)
        set_tests_properties(drm-unit drm-probe-help PROPERTIES LABELS unit TIMEOUT 15)
    endif()
endif()
if(BUILD_TESTING AND NOT CMAKE_CROSSCOMPILING)
    add_test(NAME drm-probe-cli COMMAND ${CMAKE_COMMAND}
        -DPROBE=$<TARGET_FILE:zenith_drm_probe> -P ${PROJECT_SOURCE_DIR}/tests/probe_cli.cmake)
    set_tests_properties(drm-probe-cli PROPERTIES LABELS unit TIMEOUT 15)
endif()
